function extraerMontoSinIVA(texto) { // Limpiar espacios dobles o saltos de línea raros const textoLimpio = texto.replace(/\s+/g, ' '); // 1. RegEx mejorada para buscar "Subtotal" seguido de números con formato de moneda (ej. 445,958.83) const regexSubtotal = /Subtotal\s*[:\$]?\s*([\d{1,3}(?:,\d{3})*]+(?:\.\d{2})?)/i; const coincidencia = textoLimpio.match(regexSubtotal); if (coincidencia && coincidencia[1]) { let numeroStr = coincidencia[1].replace(/,/g, ''); let valor = parseFloat(numeroStr); if (!isNaN(valor) && valor > 0) { return valor; } } // 2. Si es un PDF escaneado (imagen pura) o el formato no tiene capa de texto accesible, // permite ingresar o corregir el monto manualmente const entradaManual = prompt( 'No se pudo extraer automáticamente el Subtotal del PDF.\nPor favor ingresa el monto sin IVA:', '0.00' ); let valorManual = parseFloat(entradaManual); return isNaN(valorManual) ? 0 : valorManual; }
